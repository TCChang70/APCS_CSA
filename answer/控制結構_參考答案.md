1. **B) `&&`**

2. **B) 至少會執行一次**

3. **B) `str1.equals(str2)`**
   - 解析：在 Java 中比較字串內容必須使用 `.equals()` 方法

4. **B) `continue`**
   - 解析：`continue` 跳過當前迴圈剩餘部分，進入下一次迭代

5. **B) 只會結束最內層的迴圈**
   - 解析：`break` 只會結束包含它的最內層迴圈

6. **C) `if (x == 5)`**
   - 解析：條件判斷使用 `==`，且需要括號包圍條件

import java.util.Scanner;

public class GradeSystem {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        System.out.println("=== 學生成績等級判定系統 ===");
        
        int score;
        
        // 輸入驗證迴圈
        do {
            System.out.print("請輸入學生成績 (0-100)：");
            score = scanner.nextInt();
            
            if (score < 0 || score > 100) {
                System.out.println("❌ 成績必須在 0-100 範圍內，請重新輸入！");
            }
        } while (score < 0 || score > 100);
        
        // 等級判定
        String grade;
        String suggestion;
        
        if (score >= 90) {
            grade = "A+";
            suggestion = "優秀！請保持這個水準，可以挑戰更高難度的課程。";
        } else if (score >= 80) {
            grade = "A";
            suggestion = "良好！再努力一點就能達到優秀等級。";
        } else if (score >= 70) {
            grade = "B";
            suggestion = "普通，需要加強學習，建議多做練習題。";
        } else if (score >= 60) {
            grade = "C";
            suggestion = "及格，但成績偏低，建議尋求老師或同學的幫助。";
        } else {
            grade = "F";
            suggestion = "不及格，建議重新學習基礎概念，並考慮補考。";
        }
        
        // 輸出結果
        System.out.println("\n=== 成績評定結果 ===");
        System.out.println("學生成績：" + score + " 分");
        System.out.println("等級：" + grade);
        System.out.println("建議：" + suggestion);
        
        scanner.close();
    }
}

### **題目 4：ATM 提款機模擬系統完整解答**

```java
import java.util.Scanner;

public class ATMSystem {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        // 帳戶設定
        int balance = 50000;  // 初始餘額
        String correctPassword = "1234";
        int maxAttempts = 3;
        int attempts = 0;
        boolean isLocked = false;
        
        System.out.println("=== ATM 系統 ===");
        
        // 密碼驗證
        while (attempts < maxAttempts && !isLocked) {
            System.out.print("請輸入密碼：");
            String inputPassword = scanner.next();
            attempts++;
            
            if (inputPassword.equals(correctPassword)) {
                System.out.println("✅ 密碼正確！");
                break;
            } else {
                if (attempts < maxAttempts) {
                    System.out.println("❌ 密碼錯誤！還有 " + (maxAttempts - attempts) + " 次機會。");
                } else {
                    System.out.println("❌ 密碼錯誤次數過多，帳戶已鎖定！");
                    isLocked = true;
                }
            }
        }
        
        // 如果帳戶被鎖定，結束程式
        if (isLocked) {
            scanner.close();
            return;
        }
        
        // ATM 功能選單
        int choice;
        do {
            System.out.println("\n請選擇功能：");
            System.out.println("1. 查詢餘額");
            System.out.println("2. 提款");
            System.out.println("3. 存款");
            System.out.println("4. 離開");
            System.out.print("請輸入選項 (1-4)：");
            
            choice = scanner.nextInt();
            
            switch (choice) {
                case 1:
                    // 查詢餘額
                    System.out.println("目前餘額：" + balance + " 元");
                    break;
                    
                case 2:
                    // 提款
                    System.out.print("請輸入提款金額：");
                    int withdrawAmount = scanner.nextInt();
                    
                    if (withdrawAmount <= 0) {
                        System.out.println("❌ 提款金額必須大於 0！");
                    } else if (withdrawAmount > 20000) {
                        System.out.println("❌ 單次提款金額不能超過 20,000 元！");
                    } else if (withdrawAmount > balance) {
                        System.out.println("❌ 餘額不足！目前餘額：" + balance + " 元");
                    } else {
                        balance -= withdrawAmount;
                        System.out.println("✅ 提款成功！提取金額：" + withdrawAmount + " 元");
                        System.out.println("剩餘餘額：" + balance + " 元");
                    }
                    break;
                    
                case 3:
                    // 存款
                    System.out.print("請輸入存款金額：");
                    int depositAmount = scanner.nextInt();
                    
                    if (depositAmount <= 0) {
                        System.out.println("❌ 存款金額必須大於 0！");
                    } else {
                        balance += depositAmount;
                        System.out.println("✅ 存款成功！存入金額：" + depositAmount + " 元");
                        System.out.println("目前餘額：" + balance + " 元");
                    }
                    break;
                    
                case 4:
                    System.out.println("感謝使用，再見！");
                    break;
                    
                default:
                    System.out.println("❌ 無效選項，請重新選擇！");
            }
            
        } while (choice != 4);
        
        scanner.close();
    }
}
```

import java.util.Scanner;

public class PatternGenerator {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        int choice;
        do {
            // 顯示選單
            System.out.println("\n=== 數字圖案產生器 ===");
            System.out.println("1. 數字三角形");
            System.out.println("2. 數字金字塔");
            System.out.println("3. 乘法表");
            System.out.println("4. 離開程式");
            System.out.print("請選擇功能 (1-4)：");
            choice = scanner.nextInt();
            
            switch (choice) {
                case 1:
                    generateNumberTriangle(scanner);
                    break;
                case 2:
                    generateNumberPyramid(scanner);
                    break;
                case 3:
                    generateMultiplicationTable(scanner);
                    break;
                case 4:
                    System.out.println("感謝使用，再見！");
                    break;
                default:
                    System.out.println("❌ 無效選項，請重新選擇！");
            }
            
        } while (choice != 4);
        
        scanner.close();
    }
    
    // 數字三角形
    public static void generateNumberTriangle(Scanner scanner) {
        System.out.print("請輸入三角形層數：");
        int levels = scanner.nextInt();
        
        System.out.println("\n=== 數字三角形 ===");
        for (int i = 1; i <= levels; i++) {
            for (int j = 1; j <= i; j++) {
                System.out.print(j + " ");
            }
            System.out.println();
        }
    }
    
    // 數字金字塔
    public static void generateNumberPyramid(Scanner scanner) {
        System.out.print("請輸入金字塔層數：");
        int levels = scanner.nextInt();
        
        System.out.println("\n=== 數字金字塔 ===");
        for (int i = 1; i <= levels; i++) {
            // 印前導空格
            for (int j = 1; j <= levels - i; j++) {
                System.out.print(" ");
            }
            
            // 印數字
            for (int j = 1; j <= i; j++) {
                System.out.print(j + " ");
            }
            
            System.out.println();
        }
    }
    
    // 乘法表
    public static void generateMultiplicationTable(Scanner scanner) {
        System.out.print("請輸入乘法表範圍：");
        int range = scanner.nextInt();
        
        System.out.println("\n=== " + range + " x " + range + " 乘法表 ===");
        for (int i = 1; i <= range; i++) {
            for (int j = 1; j <= range; j++) {
                System.out.printf("%d x %d = %-3d ", i, j, i * j);
            }
            System.out.println();
        }
    }
}

public class BuggyProgram {

        for (int j = 1; j <= n; j++) {
            if (j % 2 == 0) {  // 修正：使用 == 比較

            }
        }
        System.out.println();  // 換行
        
        // 修正 5: 密碼驗證（密碼：java）

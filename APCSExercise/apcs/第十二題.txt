import java.util.*;

public class TreeHeight {
    static List<List<Integer>> tree = new ArrayList<>();
    static int[] heights;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        heights = new int[n + 1];
        boolean[] hasParent = new boolean[n + 1];

        for (int i = 0; i <= n; i++) tree.add(new ArrayList<>());

        for (int i = 1; i <= n; i++) {
            int k = sc.nextInt();
            for (int j = 0; j < k; j++) {
                int child = sc.nextInt();
                tree.get(i).add(child);
                hasParent[child] = true;
            }
        }

        int root = 1;
        for (int i = 1; i <= n; i++) {
            if (!hasParent[i]) {
                root = i;
                break;
            }
        }

        dfs(root);
        int total = 0;
        for (int i = 1; i <= n; i++) total += heights[i];

        System.out.println(root);
        System.out.println(total);
        sc.close();
    }

    static int dfs(int node) {
        if (tree.get(node).isEmpty()) {
            heights[node] = 0;
            return 0;
        }

        int maxH = 0;
        for (int child : tree.get(node)) {
            maxH = Math.max(maxH, dfs(child));
        }
        heights[node] = maxH + 1;
        return heights[node];
    }
}

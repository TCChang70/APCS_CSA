import java.util.*;

public class SpiralMatrix {
    static final int[] dx = {0, -1, 0, 1}; // 左、上、右、下
    static final int[] dy = {-1, 0, 1, 0};

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();  // N x N
        int dir = sc.nextInt(); // 初始方向

        int[][] mat = new int[n][n];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                mat[i][j] = sc.nextInt();

        int x = n / 2, y = n / 2; // 從中心開始
        StringBuilder sb = new StringBuilder();
        sb.append(mat[x][y]);

        int step = 1, count = 0;
        while (x >= 0 && y >= 0 && x < n && y < n) {
            for (int d = 0; d < 2; d++) {
                for (int i = 0; i < step; i++) {
                    x += dx[dir];
                    y += dy[dir];
                    if (x < 0 || y < 0 || x >= n || y >= n) break;
                    sb.append(mat[x][y]);
                }
                dir = (dir + 1) % 4;
            }
            step++;
            count++;
            if (count >= n) break;
        }

        System.out.println(sb.toString());
        sc.close();
    }
}

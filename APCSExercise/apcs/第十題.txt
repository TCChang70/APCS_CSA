import java.util.*;

public class BombGame {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt(); // 人數
        int m = sc.nextInt(); // 傳遞間隔
        int k = sc.nextInt(); // 爆炸次數

        LinkedList<Integer> circle = new LinkedList<>();
        for (int i = 1; i <= n; i++) circle.add(i);

        int index = 0;
        for (int i = 0; i < k; i++) {
            index = (index + m - 1) % circle.size();
            circle.remove(index);
            if (index == circle.size()) index = 0;
        }

        System.out.println(circle.get(index));
        sc.close();
    }
}

import java.util.*;

public class BaseStation {
    static int N, K;
    static int[] pos;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        N = sc.nextInt();
        K = sc.nextInt();
        pos = new int[N];
        for (int i = 0; i < N; i++) pos[i] = sc.nextInt();
        Arrays.sort(pos);

        int left = 1, right = 2_000_000_000;
        while (left < right) {
            int mid = (left + right) / 2;
            if (canCover(mid)) right = mid;
            else left = mid + 1;
        }

        System.out.println(left);
        sc.close();
    }

    static boolean canCover(int d) {
        int used = 0;
        int i = 0;
        while (i < N && used < K) {
            int cover = pos[i] + d;
            used++;
            while (i < N && pos[i] <= cover) i++;
        }
        return i == N;
    }
}

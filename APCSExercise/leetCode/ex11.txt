public class RegexMatching {
    public boolean isMatch(String s, String p) {
        int m = s.length(), n = p.length();
        boolean[][] dp = new boolean[m + 1][n + 1];
        dp[0][0] = true; // 空字串配空模式為 true

        // 處理 pattern 開頭含有 '*' 的情況，如 a*, a*b*, ...
        for (int j = 2; j <= n; j++) {
            if (p.charAt(j - 1) == '*') {
                dp[0][j] = dp[0][j - 2]; // * 當作 0 次前項
            }
        }

        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= n; j++) {
                char sc = s.charAt(i - 1);
                char pc = p.charAt(j - 1);

                if (pc == '.' || pc == sc) {
                    dp[i][j] = dp[i - 1][j - 1];
                } else if (pc == '*') {
                    char prev = p.charAt(j - 2);
                    dp[i][j] = dp[i][j - 2]; // * 當 0 次前項
                    if (prev == '.' || prev == sc) {
                        dp[i][j] = dp[i][j] || dp[i - 1][j]; // 多次前項
                    }
                }
            }
        }

        return dp[m][n];
    }

    public static void main(String[] args) {
        RegexMatching solution = new RegexMatching();

        System.out.println(solution.isMatch("aa", "a"));    // false
        System.out.println(solution.isMatch("aa", "a*"));   // true
        System.out.println(solution.isMatch("ab", ".*"));   // true
        System.out.println(solution.isMatch("mississippi", "mis*is*p*.")); // false
        System.out.println(solution.isMatch("aab", "c*a*b")); // true
    }
}
